# Match — Fase 2 (baralho + swipe + match) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development ou superpowers:executing-plans. Steps usam checkbox (`- [ ]`).

**Goal:** O coração do Tinder astrológico — baralho de cards (filtros + ordem por sinastria), swipe like/pass com likes privados, e Match (like recíproco) que cria conexão aceita, revela WhatsApp e notifica os dois.

**Architecture:** Backend estende `discovery.js` com `deck`/`swipe`/`my-matches`; nova coleção `swipes/{from}_{to}` (catch-all `if false`); o match reusa `connections` (status accepted, origin match) e o WhatsApp por duplo consentimento já existente. Frontend adiciona o baralho (`DiscoverDeck`) na seção Descobrir e a lista de Matches na seção Match.

**Tech Stack:** Node (Firebase Admin), React Native + TS + Expo, Vitest/node asserts, engine `matchScore`/`aspectsOf` (`computeAllSynastryAspects`).

**Spec:** [docs/superpowers/specs/2026-08-28-rede-tinder-astrologico-design.md](../specs/2026-08-28-rede-tinder-astrologico-design.md) · **Fase 1:** [2026-08-28-match-fase1.md](./2026-08-28-match-fase1.md) (no ar)

## Global Constraints

- **Privacidade:** `swipes` nunca lido pelo cliente (likes privados); o outro só sabe do like quando vira Match. `matchPositions`/nascimento nunca saem ao cliente.
- **Gate:** `entitlements(uid).isPremium` (ativo+trial) em `deck`/`swipe`/`my-matches` (402 se grátis).
- **Regras Firestore:** NÃO alterar (`swipes`/`connections` já `if false`).
- **Custo:** `deck` pré-filtra barato (cidade/elemento/idade) ANTES do cálculo de sinastria; cap 300; paginação.
- **i18n:** 4 idiomas; es-ES sem tildes, it-IT sem acentos.
- **Commits:** Conventional, `--no-verify`, footer Co-Authored-By. Backend `npm run deploy`.

```js
'use strict'
const PLANET_PT = { Sol: 'Sol', Lua: 'Lua', Mercurio: 'Mercúrio', Venus: 'Vênus', Marte: 'Marte' } // fallback: usa o próprio
function isReciprocalLike(mineToThem, themToMe) {
  return !!(mineToThem && themToMe && mineToThem.action === 'like' && themToMe.action === 'like')
}
function commonInterests(a = [], b = []) {
  const setB = new Set(Array.isArray(b) ? b : [])
  return (Array.isArray(a) ? a : []).filter((x) => setB.has(x)).slice(0, 6)
}
function matchReasons(aspects = [], limit = 3) {
  return [...(aspects || [])]
    .sort((x, y) => (x.orb || 0) - (y.orb || 0))
    .slice(0, limit)
    .map((a) => `${a.mine} ${a.labelPt} ${a.theirs}`)
}
function passesFilters(p = {}, f = {}) {
  if (!f) return true
  if (f.city && p.city !== f.city) return false
  if (f.element && p.element !== f.element) return false
  if (typeof f.minAge === 'number' && (typeof p.age !== 'number' || p.age < f.minAge)) return false
  if (typeof f.maxAge === 'number' && (typeof p.age !== 'number' || p.age > f.maxAge)) return false
  if (Array.isArray(f.interests) && f.interests.length) {
    const has = commonInterests(p.interests || [], f.interests).length > 0
    if (!has) return false
  }
  return true
}
module.exports = { isReciprocalLike, commonInterests, matchReasons, passesFilters, PLANET_PT }
```

```js
if (action === 'deck') {
  const ent = await entitlements(uid)
  if (!ent.isPremium) return res.status(402).json({ ok: false, gated: true })
  const { passesFilters, matchReasons, commonInterests } = require('../discovery/deck')
  const meSnap = await db.collection('users').doc(uid).get()
  let myNatal = {}; try { myNatal = (await calc(meSnap.data() || {})) || {} } catch { myNatal = {} }
  const myPos = toMatchPositions(myNatal.planets || [], EN_BY_PT)
  const meProfSnap = await db.collection(COL).doc(uid).get()
  const myInterests = (meProfSnap.exists && Array.isArray(meProfSnap.data().interests)) ? meProfSnap.data().interests : []

  // Já swipados: ids dos docs swipes/{me}_*
  const swipedSnap = await db.collection('swipes').where('from', '==', uid).limit(500).get()
  const swiped = new Set(); swipedSnap.forEach((d) => swiped.add(d.data().to))
  // Conexões existentes (accepted/blocked) — não reaparecem no baralho
  const connSnap = await db.collection('connections').where('participants', 'array-contains', uid).get()
  connSnap.forEach((d) => { const o = (d.data().participants || []).find((p) => p !== uid); if (o && ['accepted', 'blocked'].includes(d.data().status)) swiped.add(o) })

  const filters = body.filters || {}
  const snap = await db.collection(COL).where('discoverable', '==', true).limit(300).get()
  const cards = []
  snap.forEach((d) => {
    if (d.id === uid || swiped.has(d.id)) return
    const p = d.data() || {}
    if (!passesFilters(p, filters)) return
    const pos = Array.isArray(p.matchPositions) ? p.matchPositions : []
    if (!pos.length || !myPos.length) return
    const asp = aspectsOf(myPos, pos)
    cards.push({ ...publicView(d.id, p), score: matchScore(asp), reasons: matchReasons(asp), common: commonInterests(myInterests, p.interests || []) })
  })
  cards.sort((a, b) => b.score - a.score)
  return res.json({ ok: true, cards: cards.slice(0, Math.min(Number(body.limit) || 10, 20)) })
}
```

```js
if (action === 'swipe') {
  const ent = await entitlements(uid)
  if (!ent.isPremium) return res.status(402).json({ ok: false, gated: true })
  const target = str(body.uid)
  const act = body.action === 'like' ? 'like' : 'pass'
  if (!target || target === uid) return res.status(400).json({ ok: false, error: 'invalid_target' })
  const { isReciprocalLike } = require('../discovery/deck')
  const { connectionId } = require('../connections/model')
  await db.collection('swipes').doc(`${uid}_${target}`).set({ from: uid, to: target, action: act, at: FieldValue.serverTimestamp() }, { merge: true })
  let matched = false
  if (act === 'like') {
    const back = await db.collection('swipes').doc(`${target}_${uid}`).get()
    if (back.exists && isReciprocalLike({ action: 'like' }, back.data())) {
      matched = true
      const cid = connectionId(uid, target)
      await db.collection('connections').doc(cid).set({
        participants: [uid, target].sort(), requestedBy: uid, status: 'accepted',
        origin: 'match', originGroupId: null, whatsappSharedBy: [],
        createdAt: FieldValue.serverTimestamp(), acceptedAt: FieldValue.serverTimestamp(),
      }, { merge: true })
      const meName = (await db.collection('users').doc(uid).get()).data()?.name || 'Alguém'
      const themName = (await db.collection('users').doc(target).get()).data()?.name || 'Alguém'
      await db.collection('notifications').add({ userId: target, title: 'Deu Match! 💘', body: `Você e ${meName} combinaram.`, type: 'match', createdAt: FieldValue.serverTimestamp() })
      await db.collection('notifications').add({ userId: uid, title: 'Deu Match! 💘', body: `Você e ${themName} combinaram.`, type: 'match', createdAt: FieldValue.serverTimestamp() })
    }
  }
  return res.json({ ok: true, matched })
}
```
